Core SQL Queries // SQLite engine processing samples

-- Query 1: Products with highest batch failure rates SELECT b.product_name, COUNT(DISTINCT b.batch_id) AS total_batches, SUM(CASE WHEN t.overall_status='FAIL' THEN 1 ELSE 0 END) AS failures, ROUND(failures * 100.0 / total_batches, 1) AS fail_rate_pct FROM batches b LEFT JOIN test_results t ON b.batch_id = t.batch_id GROUP BY b.product_name ORDER BY fail_rate_pct DESC;
-- Query 2: Identify repeat deviations — ALCOA+ compliance risk SELECT deviation_type, severity, COUNT(*) AS total_freq, SUM(CASE WHEN is_repeat=1 THEN 1 ELSE 0 END) AS repeat_count, ROUND(repeat_count * 100.0 / total_freq, 1) AS repeat_rate_pct FROM deviations GROUP BY deviation_type, severity ORDER BY repeat_rate_pct DESC;
-- Query 3: CAPA effectiveness analysis by root cause SELECT d.root_cause, COUNT(c.capa_id) AS total_capas, SUM(CASE WHEN c.effectiveness='Effective' THEN 1 ELSE 0 END) AS effective_count, ROUND(effective_count * 100.0 / total_capas, 1) AS effectiveness_rate FROM deviations d JOIN capa_actions c ON d.deviation_id = c.deviation_id GROUP BY d.root_cause ORDER BY effectiveness_rate DESC;

Model Code // Python · scikit-learn pipeline layer

## Step 1: Load features from SQL database import pandas as pd import sqlite3 from sklearn.ensemble import RandomForestClassifier from sklearn.model_selection import train_test_split from sklearn.metrics import classification_report conn = sqlite3.connect('qc_database.db') df = pd.read_sql(""" SELECT m.storage_temp_c, m.storage_humidity_pct, m.moisture_content, m.storage_days, CASE WHEN m.tpc_status='FAIL' THEN 1 ELSE 0 END AS contaminated FROM microbial_results m """, conn) ## Step 2: Train Random Forest X = df[['storage_temp_c', 'storage_humidity_pct', 'moisture_content', 'storage_days']] y = df['contaminated'] X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2) clf = RandomForestClassifier(n_estimators=100, random_state=42) clf.fit(X_train, y_train) ## Step 3: Predict contamination risk for new batch new_batch = [[30.0, 75.0, 5.5, 60]] # temp, humidity, moisture, days risk_prob = clf.predict_proba(new_batch)[0][1] print(f"Contamination Risk: {risk_prob:.1%}") # → 88.0% ## Step 4: Feature importance evaluation outputs importance = dict(zip(X.columns, clf.feature_importances_)) # storage_temp_c: 42.6% ← most critical factor # moisture_content: 32.0% # storage_humidity_pct: 17.8% # storage_days: 7.5%

Data Integrity SQL Queries // automated violation detection engine

-- ALCOA Check 1: Missing mandatory lab entries (Attributable + Legible) SELECT 'Missing test_date' AS violation_type, COUNT(*) AS count FROM test_results WHERE test_date IS NULL UNION ALL SELECT 'Missing analyst ID', COUNT(*) FROM test_results WHERE tested_by IS NULL; -- ALCOA Check 2: Contemporaneous — detect backdated modifications SELECT * FROM lab_logs WHERE modified_time < created_time -- impossible unless backdated OR julianday(modified_time) - julianday(created_time) > 2; -- suspicious lag -- ALCOA Check 3: Accurate + Complete — open deviations aging beyond limit SELECT deviation_id, deviation_type, severity, detection_date, ROUND(julianday('now') - julianday(detection_date)) AS days_open FROM deviations WHERE status = 'Open' AND julianday('now') - julianday(detection_date) > 30 ORDER BY days_open DESC; -- ALCOA Check 4: Original — detect overwritten or overdue CAPAs SELECT capa_id, deviation_id, due_date, status, ROUND(julianday('now') - julianday(due_date)) AS days_overdue FROM capa_actions WHERE status != 'Closed' AND julianday('now') > julianday(due_date) ORDER BY days_overdue DESC;
